# 01 编辑核心 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 Tauri+Vue 脚手架之上实现 MarkWell 编辑器本体——Milkdown Crepe 挂载、18 个语法功能点（E1-E16、E19-E21）、快捷键体系与跨模块接口（实例管理/事件桥/revealRange/onUpload/keymap 注册表）。

**Architecture:** 纯前端模块（无 Rust 改动）。`@milkdown/crepe` 7.22.1 提供基座（commonmark+GFM+12 Feature），自定义代码只做四类事：keymap 注册（E2/E4/E7/E8/E19）、输入规则（E8 建表/E19 Pandoc）、自研子模块（E11 Front Matter/E12 TOC/E21 Mermaid/E20 HTML 安全）、对外接口服务（editor-manager/editor-events/reveal-range）。UI 样式按 `design-system/markwell/pages/editor.md` 在 Crepe 官方主题 CSS 之上追加变量覆盖层（亮色默认 + 暗色待 08 接入）。

**Tech Stack:** Vue 3.5 + TypeScript + Vite 6 + Vitest 4（jsdom）+ @testing-library/vue + @milkdown/crepe|kit|vue 7.22.1（三者同版本）+ KaTeX(mhchem) + mermaid（懒加载 chunk）+ @tauri-apps/plugin-opener（已装，Rust 侧已注册 opener:default）。

## Global Constraints

- 依赖版本锁定：`@milkdown/crepe`/`@milkdown/kit`/`@milkdown/vue` 一律 `7.22.1`（精确版本，禁 `^`）；禁用 `@milkdown/plugin-diagram`（已弃用，E21 走自研 renderPreview）。

- 快捷键归属：Ctrl+W/Ctrl+Tab 不注册（留给 12 窗口外壳）；编辑器内只注册 spec 第 2 节快捷键体系表中的键。
- 死代码零容忍：本次改动产生的废弃代码（脚手架 greet 命令、被替换的 smoke 断言）同次提交清理。

// src/test/editor-test-utils.ts
// 编辑器行为测试助手：直连 Crepe 实例，模拟输入与按键（@testing-library 官方测试模式的封装）
import { Crepe, CrepeFeature } from "@milkdown/crepe";
import { editorViewCtx, type Editor } from "@milkdown/kit/core";
import { insert } from "@milkdown/kit/utils";
import { TextSelection } from "@milkdown/kit/prose/state";
import type { EditorView } from "@milkdown/kit/prose/view";
import { fireEvent } from "@testing-library/dom";

/** 按键修饰键组合 */
export interface KeyMods {
  ctrl?: boolean;
  shift?: boolean;
  alt?: boolean;
  meta?: boolean;
}

/** 测试用编辑器实例句柄 */
export interface TestEditor {
  editor: Editor;
  crepe: Crepe;
  view: EditorView;
  getMarkdown(): string;
  insertText(text: string): void;
  press(key: string, mods?: KeyMods): void;
  setSelection(from: number, to: number): void;
}

/** makeTestEditor 可选配置 */
export interface MakeTestEditorOptions {
  /** 图片上传回调（对应 Crepe ImageBlock 的 onUpload，E15 用） */
  onUpload?: (file: File) => Promise<string>;
  /** 自定义 Crepe 特性开关（如关闭某 Feature 测试降级路径） */
  features?: Partial<Record<CrepeFeature, boolean>>;
}

/** 本次测试创建的实例清单，afterEach 统一销毁 */
const liveEditors: Crepe[] = [];

/** 销毁全部存活的测试编辑器（setup.ts 的 afterEach 调用） */
export function destroyTestEditors(): void {
  for (const crepe of liveEditors.splice(0)) {
    try {
      crepe.destroy();
    } catch {
      // 销毁失败不阻塞用例收尾
    }
  }
}

/**
 * 创建直连 Crepe 的测试编辑器
 * @param markdown 初始文档内容（默认空）
 * @param options 可选配置
 */
export async function makeTestEditor(
  markdown = "",
  options: MakeTestEditorOptions = {},
): Promise<TestEditor> {
  const root = document.createElement("div");
  document.body.appendChild(root);

  const crepe = new Crepe({
    root,
    defaultValue: markdown,
    features: options.features,
    featureConfigs: options.onUpload
      ? { [CrepeFeature.ImageBlock]: { onUpload: options.onUpload } }
      : undefined,
  });
  await crepe.create();
  liveEditors.push(crepe);

  const editor = crepe.editor;
  const view = editor.action((ctx) => ctx.get(editorViewCtx));

  return {
    editor,
    crepe,
    view,
    getMarkdown: () => crepe.getMarkdown(),
    insertText(text: string) {
      view.focus();
      editor.action(insert(text));
    },
    press(key: string, mods: KeyMods = {}) {
      // 通过 ProseMirror 的 DOM keydown 监听链路触发 keymap
      fireEvent.keyDown(view.dom, {
        key,
        ctrlKey: mods.ctrl ?? false,
        shiftKey: mods.shift ?? false,
        altKey: mods.alt ?? false,
        metaKey: mods.meta ?? false,
      });
    },
    setSelection(from: number, to: number) {
      view.dispatch(view.state.tr.setSelection(TextSelection.create(view.state.doc, from, to)));
    },
  };
}
```

// src/features/editor/create-editor.ts
// Crepe 工厂：统一组装 featureConfigs 与 keymap 注册注入点（10 模块扩展入口）
import { Crepe, type CrepeConfig } from "@milkdown/crepe";
// mhchem 副作用导入：KaTeX 化学式扩展（E7），全应用只需一次
import "katex/contrib/mhchem";

/** 编辑器工厂可选配置 */
export interface MarkwellEditorOptions {
  /** 图片上传回调（07 图片模块注入实现；缺省时 Crepe 自动回落 blob URL） */
  onUpload?: (file: File) => Promise<string>;
  /** 附加 Crepe 配置（keymap/插件注入点，10 设置快捷键模块使用） */
  crepeConfig?: CrepeConfig;
}

/**
 * 创建 MarkWell 编辑器实例（Crepe 7.22.1）
 * 注意：仅构造，调用方需自行 create()（@milkdown/vue 集成层会自动 create）
 * @param root 编辑器挂载根元素（由 @milkdown/vue 集成层传入）
 * @param defaultValue 初始文档内容
 * @param options 可选配置
 */
export function createMarkwellEditor(
  root: HTMLElement,
  defaultValue: string,
  options: MarkwellEditorOptions = {},
): Crepe {
  const crepe = new Crepe({
    root,
    defaultValue,
    // 默认 features 除 TopBar/Diff 外全开（与 Typora 无菜单栏编辑器形态一致）
    ...options.crepeConfig,
    featureConfigs: {
      ...(options.crepeConfig?.featureConfigs ?? {}),
      ...(options.onUpload ? { [Crepe.Feature.ImageBlock]: { onUpload: options.onUpload } } : {}),
    },
  });
  return crepe;
}
```

// src/services/editor-manager.ts
// 编辑器实例管理服务
//
// 模块 01 对外暴露的实例管理接口，供 02 自动保存、05 大纲、06 搜索定位、
// 07 图片上传、10 设置快捷键、11 字数统计、12 标题栏脏状态消费。
//
// 当前为单实例形态（01 阶段单编辑器）；05/06 多实例需求由消费方基于本服务
// 扩展或替换（接口已按「文档级」而非「应用级」划分）。
import type { Editor } from "@milkdown/kit/core";
import { editorViewCtx } from "@milkdown/kit/core";
import type { Crepe } from "@milkdown/crepe";
import { createMarkwellEditor } from "@/features/editor/create-editor";
import { destroyEditorEvents, setupEditorEvents } from "./editor-events";

/** 文档级转换器挂载点（E11 Front Matter 使用） */
export interface DocumentTransformers {
  /** 打开文档时：原始内容 → 编辑器正文（剥离 Front Matter） */
  parse?: (doc: string) => string;
  /** 保存文档时：编辑器正文 → 落盘内容（回写 Front Matter） */
  serialize?: (body: string) => string;
}

class EditorManager {
  /** 当前 Crepe 实例（单例） */
  private crepe: Crepe | undefined;
  /** 当前编辑器底层实例 */
  private editor: Editor | undefined;
  /** 文档级转换器（默认透传） */
  private transformers: DocumentTransformers = {};

  /**
   * 创建编辑器实例（销毁旧实例后创建新实例）
   * @param doc 原始文档内容（可含 Front Matter，由 transformers.parse 剥离）
   */
  async create(doc: string): Promise<void> {
    this.destroy();
    const body = (this.transformers.parse ?? ((d: string) => d))(doc);
    const root = document.createElement("div");
    document.body.appendChild(root);
    // 挂载根元素保留引用语义：Crepe 内部持有该元素，销毁时一并移除
    this.crepe = createMarkwellEditor(root, body);
    this.editor = this.crepe.editor;
    await this.crepe.create();
    setupEditorEvents(this.crepe);
  }

  /** 销毁当前实例并解除事件绑定 */
  destroy(): void {
    if (this.crepe) {
      destroyEditorEvents();
      this.crepe.destroy();
      this.crepe = undefined;
      this.editor = undefined;
    }
  }

  /** 获取 Crepe 实例（未创建返回 undefined） */
  getCrepe(): Crepe | undefined {
    return this.crepe;
  }

  /** 获取底层 Editor 实例（未创建返回 undefined） */
  getEditor(): Editor | undefined {
    return this.editor;
  }

  /** 获取当前编辑器视图（未创建返回 undefined） */
  getView() {
    return this.editor?.action((ctx) => ctx.get(editorViewCtx));
  }

  /**
   * 全量序列化当前文档（O(n)）
   * 未创建返回空串；经 transformers.serialize 回写后返回落盘内容
   */
  getMarkdown(): string {
    if (!this.crepe) return "";
    const body = this.crepe.getMarkdown();
    return (this.transformers.serialize ?? ((b: string) => b))(body);
  }

  /** 切换编辑器只读状态 */
  setReadonly(value: boolean): void {
    this.crepe?.setReadonly(value);
  }

  /** 注册文档级转换器（E11 注入；传空对象还原透传） */
  setDocumentTransformers(transformers: DocumentTransformers): void {
    this.transformers = transformers;
  }
}

/** 全局单例（模块边界内唯一入口） */
export const editorManager = new EditorManager();
```

```ts
// src/services/editor-events.ts
// 编辑器事件桥
//
// 把 Crepe 底层 listener 事件（markdownUpdated/updated/selectionUpdated）转换为
// 带防抖语义的对外回调。防抖口径（已锁定，勿改）：
//   markdownUpdated → 300ms（供 02 自动保存，全量序列化 O(n) 必须防抖）
//   updated(doc)   → 200ms（供 05 大纲收集、11 字数统计）
//   selectionUpdated → 即时（供 05 当前标题高亮、11 选中统计）
import type { Crepe } from "@milkdown/crepe";
import type { Editor, ProseMirrorNode, Selection } from "@milkdown/kit/core";

/** 事件回调集合 */
export interface EditorEventCallbacks {
  /** 防抖 300ms 后的 markdown 全文变更 */
  onMarkdownUpdated?: (markdown: string) => void;
  /** 防抖 200ms 后的文档对象变更 */
  onDocUpdated?: (doc: ProseMirrorNode) => void;
  /** 即时选区变更 */
  onSelectionUpdated?: (selection: Selection) => void;
}

/** 当前挂载的取消函数（防抖计时器清理） */
let detachFns: Array<() => void> = [];

/** 防抖包装：返回带取消能力的包装函数 */
function debounce<T extends unknown[]>(fn: (...args: T) => void, wait: number) {
  let timer: ReturnType<typeof setTimeout> | undefined;
  const wrapped = (...args: T) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), wait);
  };
  wrapped.cancel = () => clearTimeout(timer);
  return wrapped;
}

/**
 * 将 Crepe listener 事件桥接到业务回调
 * @param crepe 目标 Crepe 实例
 * @param callbacks 回调集合
 */
export function attachEditorEvents(crepe: Crepe, callbacks: EditorEventCallbacks): void {
  crepe.on((listener) => {
    if (callbacks.onMarkdownUpdated) {
      const debounced = debounce((md: string) => callbacks.onMarkdownUpdated?.(md), 300);
      listener.markdownUpdated((_ctx, md) => debounced(md));
      detachFns.push(() => debounced.cancel());
    }
    if (callbacks.onDocUpdated) {
      const debounced = debounce((doc: ProseMirrorNode) => callbacks.onDocUpdated?.(doc), 200);
      listener.updated((_ctx, doc) => debounced(doc));
      detachFns.push(() => debounced.cancel());
    }
    if (callbacks.onSelectionUpdated) {
      listener.selectionUpdated((_ctx, selection) => callbacks.onSelectionUpdated?.(selection));
    }
  });
}

/** 解绑事件桥（编辑器销毁时调用） */
export function detachEditorEvents(): void {
  for (const fn of detachFns.splice(0)) fn();
}

/** 编辑器实例生命周期内的事件桥入口（editor-manager 调用） */
export function setupEditorEvents(crepe: Crepe): void {
  // 01 阶段无固定消费者，仅暴露挂载入口；02/05/11 接入后在此订阅
  void crepe;
}

/** 编辑器实例销毁时的清理入口（editor-manager 调用） */
export function destroyEditorEvents(): void {
  detachEditorEvents();
}
```

```ts
// src/features/editor/reveal-range.ts
// revealRange 定位接口：选中目标区间 → 滚动到可视区 → 临时高亮
// 供 06 搜索替换模块的跨文件结果定位消费
import type { Editor } from "@milkdown/kit/core";
import { editorViewCtx } from "@milkdown/kit/core";
import { TextSelection } from "@milkdown/kit/prose/state";

/** 临时高亮 CSS 类名 */
const HIGHLIGHT_CLASS = "markwell-reveal-highlight";
/** 高亮自动消退时长（毫秒） */
const HIGHLIGHT_DURATION = 1200;

/**
 * 定位并高亮文档区间
 * @param editor 目标编辑器实例
 * @param from 区间起点（文档偏移，可为 0）
 * @param to 区间终点（文档偏移，越界自动收敛）
 */
export function revealRange(editor: Editor, from: number, to: number): void {
  editor.action((ctx) => {
    const view = ctx.get(editorViewCtx);
    const docSize = view.state.doc.content.size;
    // 区间越界收敛，保证 ProseMirror 选区合法
    const safeFrom = Math.max(0, Math.min(from, docSize));
    const safeTo = Math.max(safeFrom, Math.min(to, docSize));

    view.dispatch(
      view.state.tr
        .setSelection(TextSelection.create(view.state.doc, safeFrom, safeTo))
        .scrollIntoView(),
    );

    // 临时高亮：给选区所在的 DOM 节点加类，定时移除
    const { from: selFrom, to: selTo } = view.state.selection;
    const dom = view.nodeDOM(selFrom) as HTMLElement | null;
    if (dom) {
      dom.classList.add(HIGHLIGHT_CLASS);
      setTimeout(() => dom.classList.remove(HIGHLIGHT_CLASS), HIGHLIGHT_DURATION);
    }
  });
}

- [ ] **Step 3: 实现 keymaps.ts 注册表与 Ctrl+[/Ctrl+] 反向配对**

> 注册机制已逐行核对自 @milkdown/core 7.22.1 构建产物：`KeymapManager.add({ key, priority, onRun })`，`onRun(ctx)` 返回 ProseMirror Command；build 时按 `(b.priority ?? 50)` 降序排序后 chainCommands，先返回 true 者胜。表格内置 NextCell 为 `{ priority: 100, shortcuts: ["Mod-]", "Tab"] }`，PrevCell 为 `Mod-[`/`Shift-Tab`（priority 默认 50）——因此本模块自定义键位统一用 priority 200 压制内置。

```ts
// src/features/editor/keymaps.ts
// 模块级 keymap 注册表
//
// 在 Crepe create() 之前的配置阶段调用（create-editor.ts 注入点），
// 10 设置快捷键模块通过 addEditorKeymap 扩展自定义键位。
// 注册机制：keymapCtx（KeymapManager），add 接受 { key, priority, onRun }；
// onRun(ctx) 返回 ProseMirror Command，build 时按 priority 降序 chainCommands。
import type { Ctx } from "@milkdown/kit/ctx";
import { keymapCtx } from "@milkdown/kit/core";
import { liftListItem, sinkListItem } from "@milkdown/kit/prose/schema-list";
import { listItemSchema } from "@milkdown/kit/preset/commonmark";

/** 单条 keymap 注册项 */
export interface EditorKeymapEntry {
  /** 按键组合（ProseMirror keymap 语法，如 "Mod-["） */
  key: string;
  /** 命中后执行的动作（返回 ProseMirror Command） */
  onRun: (ctx: Ctx) => (state: unknown, dispatch?: unknown, view?: unknown) => boolean;
  /** 冲突优先级（默认 200，高于表格内置的 100 与 baseKeymap 的 50） */
  priority?: number;
}

/** 已注册键位清单（供 10 模块查询冲突） */
const registry: EditorKeymapEntry[] = [];

/**
 * 注册编辑器键位（create() 前调用）
 * @param entry 键位定义
 */
export function addEditorKeymap(entry: EditorKeymapEntry): void {
  registry.push(entry);
}

/** 查询已注册键位（10 模块扩展前查重） */
export function listEditorKeymaps(): readonly EditorKeymapEntry[] {
  return registry;
}

/** 是否已注册过该键位（10 模块扩展前查重） */
export function hasEditorKeymap(key: string): boolean {
  return registry.some((e) => e.key === key);
}

/**
 * 将注册表写入 keymapCtx（create-editor.ts 的 config 阶段调用一次）
 * @param ctx milkdown 配置上下文
 */
export function applyEditorKeymaps(ctx: Ctx): void {
  for (const entry of registry) {
    ctx.get(keymapCtx).add({
      key: entry.key,
      priority: entry.priority ?? 200,
      onRun: entry.onRun,
    });
  }
}

// ── 内置键位定义（模块加载即注册，create() 时统一生效）──

// Typora 特有反向 Indent/Outdent 配对（用户实测裁决，见 01 调研第 7 节 #1）：
//   Ctrl+[ = Indent（缩进增加，sinkListItem）——与 VS Code 等业界习惯相反
//   Ctrl+] = Outdent（缩进减少，liftListItem）
// priority 200 同时压制表格内置的 Mod-[（PrevCell，priority 50/100）：
// 在列表项内 sink/lift 生效即消费按键；非列表上下文返回 false 自然回落内置行为。
addEditorKeymap({
  key: "Mod-[",
  onRun: () => sinkListItem(listItemSchema.type()),
});

addEditorKeymap({
  key: "Mod-]",
  onRun: () => liftListItem(listItemSchema.type()),
});
```

// keymaps.ts 追加（import 段补充）
import { findParentNode } from "@milkdown/kit/prose";
import type { EditorState, Transaction } from "@milkdown/kit/prose/state";
import {
  headingSchema,
  paragraphSchema,
  setBlockTypeCommand,
  wrapInHeadingCommand,
} from "@milkdown/kit/preset/commonmark";

// ── 标题级别键位（Typora 官方键位表）──
// Ctrl+1~6 设为对应级别、Ctrl+0 转段落、Ctrl+=/- 增减级别
for (let level = 1; level <= 6; level++) {
  addEditorKeymap({
    key: `Mod-${level}`,
    onRun: () => wrapInHeadingCommand(headingSchema.type(), level),
  });
}

addEditorKeymap({
  key: "Mod-0",
  onRun: () => setBlockTypeCommand(paragraphSchema.type()),
});

/**
 * 增减标题级别命令：读取光标所在 heading 级别并钳制在 1-6
 * @param delta 1 = 升级（级别数字减一）；-1 = 降级（级别数字加一）
 */
function headingLevelCommand(delta: 1 | -1) {
  return (ctx: Ctx) => {
    return (state: EditorState, dispatch?: (tr: Transaction) => void) => {
      // 找到选区所在的 heading 节点
      const parent = findParentNode((node) => node.type === headingSchema.type())(state.selection);
      if (!parent) return false; // 非标题上下文不消费按键
      const currentLevel = parent.node.attrs.level as number;
      // 级别钳制：H1 升级保持 H1、H6 降级保持 H6（AC-E2-5 越界保护）
      const nextLevel = Math.min(6, Math.max(1, currentLevel - delta));
      return wrapInHeadingCommand(headingSchema.type(), nextLevel)(state, dispatch);
    };
  };
}

addEditorKeymap({ key: "Mod-=", onRun: headingLevelCommand(1) }); // 升级（数字减小）
addEditorKeymap({ key: "Mod--", onRun: headingLevelCommand(-1) }); // 降级（数字增大）
```

（说明：`findParentNode` 从 `@milkdown/kit/prose` 导入（构建产物导入清单已见）；`wrapInHeadingCommand(type, level)` 的 level 参数与 `setBlockTypeCommand` 均为 commonmark 导出（构建产物已核实）。`Mod-=`/`Mod--` 为 prosemirror-keymap 合法键名。若实测发现 baseKeymap 的 `Mod-Alt-1~6` 与 `Mod-1~6` 在 Windows 下冲突，以实际按键事件为准——二者修饰键不同（Alt vs 无），不会冲突。）

> 已核实（@milkdown/preset-gfm 7.22.1 构建产物）：内置建表规则为 `|cols x rows|`（`insertTableInputRule`，正则 `/^\|(?<col>\d+)[xX](?<row>\d+)\|\s$/`，走 `createTable(ctx, rows, cols)`），与 Typora 式 `| 表头 | 表头 |` 不同且不填充单元格文本。本任务自建规则：解析键入行 → 用 schema 节点手工构建「表头行含输入文本」的表格。

```ts
// src/features/editor/input-rules.ts
// 模块级自定义输入规则
//
// Crepe 内置输入规则与 Typora 行为不一致时的补充（本模块：E8 Typora 式建表、
// E19 Pandoc 行内数学）。通过 registerEditorInputRules 在 create() 前注入。
import type { Ctx } from "@milkdown/kit/ctx";
import { InputRule } from "@milkdown/kit/prose/inputrules";
import { inputRulesCtx } from "@milkdown/kit/core";
import type { Node, Schema } from "@milkdown/kit/prose/model";
import { tableHeaderRowSchema, tableHeaderSchema, tableSchema } from "@milkdown/kit/preset/gfm";

/** 已注册的输入规则清单 */
const inputRules: InputRule[] = [];

/**
 * 追加自定义输入规则（create() 前调用）
 * @param rule ProseMirror 输入规则
 */
export function addEditorInputRule(rule: InputRule): void {
  inputRules.push(rule);
}

/** 查询已注册规则（测试用） */
export function listEditorInputRules(): readonly InputRule[] {
  return inputRules;
}

/**
 * Typora 式建表：解析 `| a | b |` 行内容，构建表头行含文本的表格节点
 * @param schema 编辑器 schema
 * @param cells 单元格文本列表
 */
function buildTyporaTable(schema: Schema, cells: string[]): Node {
  // 单元格：表头节点内嵌段落 + 输入文本
  const cellNodes = cells.map((text) =>
    tableHeaderSchema.type().create(null, schema.nodes.paragraph.create(null, schema.text(text))),
  );
  const headerRow = tableHeaderRowSchema.type().create(null, cellNodes);
  return tableSchema.type().create(null, headerRow);
}

/**
 * Typora 式建表输入规则：`| 表头 | 表头 |` + Enter → 创建表格
 * 规则：整行由 1~N 个竖线分隔单元格组成，行首行尾均为 `|`
 */
const typoraTableRule = new InputRule(/^\|(?:\s*[^|\n]+\s*\|)+\s*$/, (state, match, start, end) => {
  // 解析单元格文本（去掉首尾竖线后按 | 切分）
  const cells = match[0]
    .slice(1, -1)
    .split("|")
    .map((c) => c.trim());
  if (cells.length === 0) return null;
  const table = buildTyporaTable(state.schema, cells);
  // 用表格节点替换输入行，光标移入首个单元格
  const tr = state.tr.replaceRangeWith(start, end, table);
  return tr;
});

addEditorInputRule(typoraTableRule);

/**
 * 将注册的输入规则合并进内置规则（create-editor.ts 的 config 阶段调用一次）
 * @param ctx milkdown 配置上下文
 */
export function registerEditorInputRules(ctx: Ctx): void {
  ctx.update(inputRulesCtx, (rules) => [...rules, ...inputRules]);
}
```

（`buildTyporaTable` 的 schema 参数类型写为 `Schema`，从 `@milkdown/kit/prose/model` 导入 `Schema`/`Node` 类型。节点类型名以安装源码为准——`tableSchema`/`tableHeaderRowSchema`/`tableHeaderSchema` 均为 preset-gfm 导出（构建产物导出清单已核实）。create-editor.ts 在 Crepe 构造后追加 `crepe.editor.config((ctx) => registerEditorInputRules(ctx))`。）

- [ ] **Step 4: 实现 Tab 末单元格加行 keymap**

> 已核实（@milkdown/preset-gfm 7.22.1 构建产物）：不存在 `appendRowCommand`；「加行」正确命令为 `addRowAfterCommand`（内部走 `addRowWithAlignment(ctx, tr, rect, rect.bottom)`）。内置表格 Tab 键位为 NextCell `{ priority: 100, shortcuts: ["Mod-]", "Tab"] }`。任意行末格加在表格末尾：直接把选区定位到表格最后一行，用 `addRowWithAlignment` 在末行之后（rect.bottom）插入新行。

```ts
// keymaps.ts 追加（import 段补充）
import { addRowWithAlignment } from "@milkdown/kit/preset/gfm";
import { TextSelection } from "@milkdown/kit/prose/state";
import { findTable, selectedRect, TableMap } from "@milkdown/kit/prose/tables";

/**
 * 表格内 Tab 加行：任意行最后一个单元格按 Tab 在表格末尾新增一行
 * 中间单元格返回 false 放行，由内置 NextCell（priority 100）接管移格。
 * 本键位 priority 200，先于内置判定。
 */
function makeTableTabAddRowCommand() {
  return (ctx: Ctx) => {
    return (state: EditorState, dispatch?: (tr: Transaction) => void) => {
      const table = findTable(state.selection);
      if (!table) return false; // 不在表格内 → 放行（列表缩进/内置行为）
      const rect = selectedRect(state.selection);
      const map = TableMap.get(table.node);
      // 当前单元格是否为本行最后一个单元格（TableMap 行优先编号）
      const isLastCellOfRow = (rect.map + 1) % map.width === 0;
      if (!isLastCellOfRow) return false; // 中间单元格 → 放行给内置移格
      if (!dispatch) return true;
      // 任意行末格：把选区定位到表格最后一行末格，再执行末尾加行
      const lastRowEnd = table.start + table.node.content.size;
      const lastRowSelection = TextSelection.create(
        state.doc,
        Math.max(table.start + 1, lastRowEnd - map.width),
      );
      const lastRowRect = selectedRect(lastRowSelection);
      dispatch(addRowWithAlignment(ctx, state.tr, lastRowRect, lastRowRect.bottom));
      return true;
    };
  };
}

addEditorKeymap({ key: "Tab", onRun: makeTableTabAddRowCommand() });
```

（说明：`findTable`/`selectedRect`/`TableMap` 均来自 `@milkdown/kit/prose/tables`（构建产物导入清单已核实）；`EditorState`/`Transaction` 类型从 `@milkdown/kit/prose/state` 导入。`lastRowSelection` 的定位常数以实测修正——意图明确：选中最后一行内任意单元格使 `selectedRect` 覆盖末行，`rect.bottom` 即表格末尾插入点。若 `addRowWithAlignment` 直接可用且行为一致，可简化为对末行选区调用；两种形态二选一，验收以 AC-E8-3 断言为准。）

// src/features/editor/footnote-tooltip.ts
// 脚注悬停预览：引用处悬停弹出定义内容（E9）
//
// 机制：plugin-tooltip 的 tooltipFactory 注册一个脚注专用浮层——
// 悬停 footnoteReference 节点时显示对应 footnoteDefinition 的内容。
// 在 create-editor.ts 中追加该插件。
import { $prose } from "@milkdown/kit/utils";
import { tooltipFactory } from "@milkdown/kit/plugin/tooltip";
import { footnoteDefinitionSchema, footnoteReferenceSchema } from "@milkdown/kit/preset/gfm";
import { PluginKey } from "@milkdown/kit/prose/state";

/** 脚注预览浮层 key */
const footnoteTooltipKey = new PluginKey("MILKDOWN_FOOTNOTE_TOOLTIP");

/**
 * 创建脚注悬停预览浮层
 * 悬停命中 footnoteReference 时，从文档中查找同 label 的 footnoteDefinition 渲染内容
 */
export const footnoteTooltip = $prose(() => {
  const tooltip = tooltipFactory(footnoteTooltipKey.key);

  tooltip.shouldShow = (view) => {
    const { doc, selection } = view.state;
    const { from } = selection;
    // 选区落在脚注引用节点内才显示
    const ref = doc.nodeAt(from);
    const parent = doc.resolve(from).parent;
    return (
      (ref?.type === footnoteReferenceSchema.type() ||
        parent?.type === footnoteReferenceSchema.type()) &&
      from <= doc.content.size
    );
  };

  tooltip.update = (view) => {
    const { doc, selection } = view.state;
    const { from } = selection;
    // 查找引用的 label 与对应定义
    const node = doc.nodeAt(from) ?? doc.resolve(from).parent;
    const label = node?.attrs?.label as string | undefined;
    if (!label) return;
    const def = doc.descendants(
      (n) => n.type === footnoteDefinitionSchema.type() && n.attrs.label === label,
    );
    const content = def?.textContent ?? "(无定义)";
    tooltip.render(view, {
      dom: createFootnoteTooltipDom(content),
    });
  };

  return tooltip.plugin;
});

/** 构建预览 DOM（浮层样式沿用 Crepe tooltip 主题类） */
function createFootnoteTooltipDom(content: string): HTMLElement {
  const el = document.createElement("div");
  el.className = "milkdown-tooltip markwell-footnote-tooltip";
  el.textContent = content;
  return el;
}
```

（`tooltipFactory` 的精确方法名（`shouldShow`/`update`/`render`/`plugin`）以 `@milkdown/kit/plugin/tooltip` 安装源码为准——link-tooltip 组件是其标准用法样板。create-editor.ts 追加 `.use(footnoteTooltip)` 或 `editor.config` 注入。）

    const mathInline = schema.nodes["math_inline"];

```ts
// src/features/editor/frontmatter/frontmatter.ts
// YAML Front Matter 处理（E11）
//
// 策略（已锁定）：加载时剥离存内存、保存/导出回写，不进文档树。
// 与 E10 水平线 `---` 的歧义按「文档第一个块」判定（AC-E11-3）。
// Typora 专有属性白名单（typora-root-url/typora-copy-images-to）供 07 图片模块读取。
/** 文首 front matter 定界符 */
const FM_DELIMITER = "---";
/** 首块正则：文档开头即 --- 且后续存在闭合 --- */
const FRONT_MATTER_RE = /^---\r?\n([\s\S]*?)\r?\n---(?:\r?\n|$)/;

/**
 * 解析文档：剥离文首 front matter
 * @param md 原始 markdown 文档
 * @returns frontMatter 为 null 表示无有效 front matter；body 为编辑器正文
 */
export function parseFrontMatter(md: string): { frontMatter: string | null; body: string } {
  const match = FRONT_MATTER_RE.exec(md);
  if (!match) return { frontMatter: null, body: md };
  const inner = match[1];
  if (!isValidFrontMatter(inner)) {
    // 非法 YAML：不剥离、按原文保留并告警（AC-E11-4）
    console.warn("[MarkWell] 文首 front matter YAML 非法，按原文保留：", inner.slice(0, 80));
    return { frontMatter: null, body: md };
  }
  return { frontMatter: inner, body: md.slice(match[0].length) };
}

/**
 * 回写 front matter（原样保留，不做 YAML 重序列化）
 * @param frontMatter 剥离时的原文
 * @param body 编辑器正文
 */
export function reinsertFrontMatter(frontMatter: string, body: string): string {
  return `${FM_DELIMITER}\n${frontMatter}\n${FM_DELIMITER}\n${body}`;
}

/**
 * 校验 front matter 内文合法性（行级最小校验）
 * 合法行：`key: value` 键值对、注释（#）、空行；括号/引号不配对视为非法
 * @param fm front matter 内文（不含定界符）
 */
export function isValidFrontMatter(fm: string): boolean {
  const lines = fm.split(/\r?\n/);
  if (lines.length === 0) return false;
  for (const line of lines) {
    const trimmed = line.trim();
    if (trimmed === "" || trimmed.startsWith("#")) continue; // 空行与注释合法
    if (!/^[\w-]+\s*:/.test(trimmed)) return false; // 非键值行非法
    // 括号/引号成对校验（防未闭合结构）
    if (!isBalanced(trimmed)) return false;
  }
  return true;
}

/** 括号/引号是否成对 */
function isBalanced(line: string): boolean {
  const pairs: Record<string, string> = { "[": "]", "{": "}", '"': '"', "'": "'" };
  const stack: string[] = [];
  for (const ch of line) {
    if (ch in pairs) stack.push(ch);
    else if (ch === "]" || ch === "}" || ch === '"' || ch === "'") {
      const open = stack.pop();
      if (!open || pairs[open] !== ch) return false;
    }
  }
  return stack.length === 0;
}

/**
 * 读取指定键的值（行级解析，支持无空格 `key:value`）
 * @param fm front matter 内文
 * @param key 目标键名
 * @returns 字符串值；不存在返回 undefined
 */
export function readFrontMatterKey(fm: string, key: string): string | undefined {
  const line = fm.split(/\r?\n/).find((l) => {
    const m = /^([\w-]+)\s*:\s*(.*)$/.exec(l.trim());
    return m?.[1] === key;
  });
  if (!line) return undefined;
  const m = /:\s*(.*)$/.exec(line);
  return m?.[1]?.trim();
}
```

// TOC 目录插件（E12）
//
// 自定义 `toc` 原子节点 + DOM NodeView：
//   - 输入规则 `[toc]` + Enter 生成
//   - 序列化落盘为 `[toc]` 文本、解析回读还原节点
//   - 文档更新后由 create-editor.ts 统一通知防抖重算（300ms）

import { editorViewCtx } from "@milkdown/kit/core";
import { nodeViewCtx } from "@milkdown/kit/core";
import { headingSchema } from "@milkdown/kit/preset/commonmark";
import { TextSelection } from "@milkdown/kit/prose/state";
import type { Node as ProseMirrorNode } from "@milkdown/kit/prose/model";
import type { EditorView } from "@milkdown/kit/prose/view";

/** toc 节点名 */
export const tocNodeName = "toc";

/** toc 节点 schema：原子块节点（无内容，渲染由 NodeView 接管） */
const tocSchema = $nodeSchema(tocNodeName, () => ({
  group: "block",
  atom: true,
  parseDOM: [{ tag: `div[data-node-type="${tocNodeName}"]` }],
  toDOM: () => ["div", { "data-node-type": tocNodeName, class: "markwell-toc" }],
  parseMarkdown: {
    match: (node) => node.type === "text" && node.value === "[toc]",
    runner: (state, _node, type) => {
      state.addNode(type);
    },
  },
  toMarkdown: {
    match: (node) => node.type.name === tocNodeName,
    runner: (state) => {
      state.addNode("text", undefined, "[toc]");
    },
  },
}));

/** `[toc]` 输入规则：整行为 [toc] 时转为目录节点 */
const tocInputRule = $inputRule(/^\[toc\]$/, (ctx) => {
  const view = ctx.get(editorViewCtx);
  const { tr, selection } = view.state;
  const tocType = view.state.schema.nodes[tocNodeName];
  if (!tocType) return false;
  // 整行替换为 toc 节点（正则锚定整行，$from.before() 即行首）
  const start = selection.$from.before();
  view.dispatch(tr.replaceRangeWith(start, selection.from, tocType.create()));
  return true;
});

```ts
// toc-plugin.ts 继续

/** 存活的 toc 节点视图清单（更新通知用） */
const liveTocViews = new Set<TocNodeView>();

/**
 * toc 节点视图：渲染目录列表 DOM
 * 目录结构来自文档 heading 节点（含层级），点击条目定位到标题
 */
class TocNodeView {
  /** 容器 DOM */
  dom: HTMLElement;

  constructor(
    private node: ProseMirrorNode,
    private view: EditorView,
    private getPos: () => number | undefined,
  ) {
    this.dom = document.createElement("div");
    this.dom.className = "markwell-toc";
    this.dom.contentEditable = "false";
    liveTocViews.add(this);
    this.rebuild();
  }

  /** 重建目录 DOM：遍历文档收集 heading 节点 */
  rebuild(): void {
    this.dom.textContent = "";
    const headings: Array<{ level: number; text: string; pos: number }> = [];
    this.view.state.doc.descendants((node, pos) => {
      if (node.type === headingSchema.type()) {
        headings.push({ level: node.attrs.level as number, text: node.textContent, pos });
      }
    });
    if (headings.length === 0) {
      // AC-E12-4：无标题时显示提示
      const empty = document.createElement("span");
      empty.className = "markwell-toc__empty";
      empty.textContent = "（无标题）";
      this.dom.appendChild(empty);
      return;
    }
    for (const h of headings) {
      const item = document.createElement("div");
      item.className = "markwell-toc__item";
      item.style.paddingLeft = `${(h.level - 1) * 16}px`;
      item.textContent = h.text;
      // AC-E12-3：点击跳转到对应标题（定位到标题文本起点并临时高亮）
      item.addEventListener("click", () => {
        const from = h.pos + 1;
        this.view.dispatch(
          this.view.state.tr
            .setSelection(TextSelection.create(this.view.state.doc, from, from + h.text.length))
            .scrollIntoView(),
        );
        const dom = this.view.nodeDOM(from) as HTMLElement | null;
        if (dom) {
          dom.classList.add("markwell-reveal-highlight");
          setTimeout(() => dom.classList.remove("markwell-reveal-highlight"), 1200);
        }
      });
      this.dom.appendChild(item);
    }
  }

  /** 销毁：退出存活清单 */
  destroy(): void {
    liveTocViews.delete(this);
  }

  /** 忽略外部 DOM 变更（原子节点） */
  ignoreMutation(): boolean {
    return true;
  }
}

/**
 * 通知全部存活 toc 视图重算（create-editor.ts 在 updated 事件防抖后调用）
 */
export function notifyTocViews(): void {
  for (const view of liveTocViews) view.rebuild();
}

/** 节点视图工厂（供 nodeViewCtx 注册） */
export function tocNodeViewFactory() {
  return (node: ProseMirrorNode, view: EditorView, getPos: () => number | undefined) =>
    new TocNodeView(node, view, getPos);
}

/** 节点视图注册配置（create-editor.ts 注入） */
export function setupTocNodeView(ctx: Ctx): void {
  ctx.update(nodeViewCtx, (prev) => [
    ...prev,
    { name: tocNodeName, factory: tocNodeViewFactory() },
  ]);
}

// E14：Ctrl+点击链接在系统浏览器打开
//
// 普通点击仍由内置 LinkTooltip 处理（展开编辑），本插件仅在
// Ctrl/Meta+点击时拦截并转交 openExternalUrl。
import { $prose } from "@milkdown/kit/utils";
import { Plugin } from "@milkdown/kit/prose/state";
import { linkSchema } from "@milkdown/kit/preset/commonmark";
import { openExternalUrl } from "@/services/external-open";

/**
 * Ctrl+点击打开链接插件
 * handleClick 在选区命中 link 标记且事件带 Ctrl/Meta 修饰时拦截
 */
export const openLinkPlugin = $prose(() => {
  return new Plugin({
    props: {
      handleClick(view, pos, event) {
        const modPressed = event.ctrlKey || event.metaKey;
        if (!modPressed) return false; // 普通点击放行给 LinkTooltip
        // 查找点击位置的 link 标记
        const $pos = view.state.doc.resolve(pos);
        const marks = $pos.marks();
        const linkMark = marks.find((m) => m.type === linkSchema.type());
        if (!linkMark) return false;
        const href = linkMark.attrs.href as string;
        if (!href) return false;
        event.preventDefault();
        void openExternalUrl(href);
        return true;
      },
    },
  });
});
```

（`linkSchema.type()` 与 mark 查找逻辑以 `@milkdown/kit/preset/commonmark` 实际导出为准；点击位置可能落在 link 内部 text 节点，`$pos.marks()` 语义由 ProseMirror 保证。create-editor.ts 追加 `.use(openLinkPlugin)`。）

```ts
// src/features/editor/html/html-sanitize.ts
// HTML 安全清洗（E20 安全路径）
//
// 策略（与 Typora 官方机制一致，见 01 调研第 2 节）：
//   - 白名单清洗：DOMPurify 默认白名单 + script/onload 等事件属性禁用
//   - iframe：允许但强制 sandbox（无本地文件访问权限）
//   - class/id/data-*：渲染时剥离、导出时保留（编辑视图与落盘分离）
import DOMPurify from "dompurify";

/** iframe 允许列表：仅 https 远程内容（本地文件访问已被 sandbox 阻断） */
const IFRAME_ALLOWED = ["iframe"];

/**
 * 清洗 HTML 片段（进入编辑视图前调用）
 * @param html 原始 HTML 字符串
 * @returns 清洗后的安全 HTML
 */
export function sanitizeHtml(html: string): string {
  return DOMPurify.sanitize(html, {
    ADD_TAGS: IFRAME_ALLOWED,
    // 强制 iframe 使用 sandbox：隔离脚本与本地文件访问（AC-E20-4）
    ADD_ATTR: ["sandbox"],
  });
}

/**
 * 渲染时剥离装饰性属性（class/id/data-*），导出时保留原文（AC-E20-2）
 * 实现：解析 DOM 后逐节点移除属性再序列化，仅用于编辑视图渲染，
 * 文档内容（getMarkdown）不做此剥离。
 * @param html 已清洗的 HTML
 * @returns 剥离后的 HTML
 */
export function stripHtmlAttrsAtRender(html: string): string {
  const doc = new DOMParser().parseFromString(html, "text/html");
  for (const el of doc.body.querySelectorAll<HTMLElement>("*")) {
    el.removeAttribute("class");
    el.removeAttribute("id");
    for (const attr of [...el.attributes]) {
      if (attr.name.startsWith("data-")) el.removeAttribute(attr.name);
    }
  }
  return doc.body.innerHTML;
}
```

````ts
// src/features/editor/mermaid/legacy-transform.ts
// legacy 图表语法 → mermaid 语法转换（E21）
//
// Typora 旧版支持两套 legacy 语法：```sequence（js-sequence-diagrams）与
// ```flow（flowchart.js）。mermaid 是其超集：sequence 语法子集可直接复用，
// flow 的 token 语法（st=>start 等）需做一次结构映射。
// 映射仅覆盖常用节点类型（start/end/operation/condition/inputoutput）。
/** legacy sequence 直接可用（mermaid sequenceDiagram 兼容其子集） */
const SEQUENCE_HEADER = "sequenceDiagram";

/**
 * legacy 图表语法转换
 * @param language 代码块语言标识
 * @param content 代码块内容
 * @returns 可直接交给 mermaid 的语法文本（非 legacy 语言原样透传）
 */
export function transformLegacyDiagram(language: string, content: string): string {
  if (content.length === 0) return content;
  const lang = language.toLowerCase();
  if (lang === "sequence") {
    // legacy sequence 语法是 mermaid sequenceDiagram 的子集（A->B: msg / Note left of A）
    return `${SEQUENCE_HEADER}\n${content}`;
  }
  if (lang === "flow") {
    return transformLegacyFlow(content);
  }
  return content; // 其余语言透传（含 mermaid 本身）
}

/** flowchart.js token 语法 → mermaid flowchart */
function transformLegacyFlow(content: string): string {
  const lines = content.split("\n");
  const out: string[] = ["flowchart TD"];
  for (const line of lines) {
    const trimmed = line.trim();
    const m = /^([\w-]+)=>(start|end|operation|condition|inputoutput):\s*(.+)$/.exec(trimmed);
    if (!m) {
      // 非节点声明行（连接线等）原样保留
      out.push(trimmed);
      continue;
    }
    const [, id, kind, label] = m;
    // mermaid 节点形态映射：start/end/inputoutput 圆角、operation 矩形、condition 菱形
    const shape = kind === "condition" ? `{${label}}` : `[${label}]`;
    out.push(`${id}${shape}`);
  }
  return out.join("\n");
}
````

// src/features/editor/mermaid/mermaid-preview.ts
// Mermaid 图表渲染（E21）
//
// 复用 CodeMirror Feature 的 renderPreview 钩子（latex feature 同款模式）：
// language === 'mermaid'（含 legacy sequence/flow）时懒加载 mermaid chunk 渲染，
// 其余语言回落到前序 renderPreview（latex 等内置渲染不受影响）。
// 懒加载：mermaid 体积大，首次渲染时动态 import（独立 chunk）。
/** 渲染中占位 HTML（applyPreview 完成前） */
const LOADING_HTML = '<div class="markwell-mermaid-loading">图表渲染中…</div>';

/**
 * 创建 mermaid 预览钩子（链式包裹前序 renderPreview）
 * @param prev 前序 renderPreview（内置 latex 等）
 */
export function createMermaidRenderPreview(
  prev: (language: string, content: string, applyPreview: (html: string) => void) => string,
) {
  return (language: string, content: string, applyPreview: (html: string) => void): string => {
    const lang = language.toLowerCase();
    if (lang === "mermaid" || lang === "sequence" || lang === "flow") {
      if (content.length === 0) return "";
      // 异步渲染：先回占位，渲染完成后 applyPreview 替换
      void renderMermaidAsync(content, lang).then(
        (html) => applyPreview(html),
        () => applyPreview(errorHtml()),
      );
      return LOADING_HTML;
    }
    return prev(language, content, applyPreview);
  };
}

/** 错误提示 HTML（AC-E21-3：非法语法显示解析错误而非崩溃） */
function errorHtml(): string {
  return '<div class="markwell-mermaid-error">Mermaid 解析错误：请检查图表语法</div>';
}

/**
 * 异步渲染 mermaid（懒加载 chunk）
 * @param content 图表语法（legacy 已转换）
 * @param lang 原始语言标识
 */
async function renderMermaidAsync(content: string, lang: string): Promise<string> {
  const { default: mermaid } = await import("mermaid");
  mermaid.initialize({ startOnLoad: false, securityLevel: "strict" });
  const source = transformLegacyDiagram(lang, content);
  const { svg } = await mermaid.render("markwell-mermaid", source);
  return `<div class="markwell-mermaid-svg">${svg}</div>`;
}

export function saveSvgAsFile(svg: string, suggestedName: string): void {

  });
}
